9. Now the code

This code does three things simultaneously:

drives both motors forward
reads both encoders
displays encoder counts

We're deliberately running at only PWM 100 initially.

// =====================================================
// ESP32 + L298N + 2 N20 ENCODER MOTORS
// MOTOR + ENCODER SIMULTANEOUS TEST
// =====================================================

// ---------------- MOTOR PINS ----------------

#define IN1 27
#define IN2 26

#define IN3 25
#define IN4 33

#define ENA 14
#define ENB 32


// ---------------- ENCODER PINS ----------------

#define LEFT_A   18
#define LEFT_B   19

#define RIGHT_A  21
#define RIGHT_B  22


// ---------------- ENCODER COUNTERS ----------------

volatile long leftCount = 0;
volatile long rightCount = 0;


// =====================================================
// LEFT ENCODER INTERRUPT
// =====================================================

void IRAM_ATTR leftEncoderISR()
{
  int A = digitalRead(LEFT_A);
  int B = digitalRead(LEFT_B);

  if (A == B)
    leftCount++;
  else
    leftCount--;
}


// =====================================================
// RIGHT ENCODER INTERRUPT
// =====================================================

void IRAM_ATTR rightEncoderISR()
{
  int A = digitalRead(RIGHT_A);
  int B = digitalRead(RIGHT_B);

  if (A == B)
    rightCount++;
  else
    rightCount--;
}


// =====================================================
// LEFT MOTOR
// =====================================================

void leftMotor(int speed)
{
  speed = constrain(speed, -255, 255);

  // Your physical forward direction was opposite
  // to our original software mapping.

  if (speed > 0)
  {
    digitalWrite(IN1, LOW);
    digitalWrite(IN2, HIGH);
  }
  else if (speed < 0)
  {
    digitalWrite(IN1, HIGH);
    digitalWrite(IN2, LOW);
  }
  else
  {
    digitalWrite(IN1, LOW);
    digitalWrite(IN2, LOW);
  }

  analogWrite(ENA, abs(speed));
}


// =====================================================
// RIGHT MOTOR
// =====================================================

void rightMotor(int speed)
{
  speed = constrain(speed, -255, 255);

  // Same physical forward convention

  if (speed > 0)
  {
    digitalWrite(IN3, LOW);
    digitalWrite(IN4, HIGH);
  }
  else if (speed < 0)
  {
    digitalWrite(IN3, HIGH);
    digitalWrite(IN4, LOW);
  }
  else
  {
    digitalWrite(IN3, LOW);
    digitalWrite(IN4, LOW);
  }

  analogWrite(ENB, abs(speed));
}


// =====================================================
// STOP
// =====================================================

void stopMotors()
{
  analogWrite(ENA, 0);
  analogWrite(ENB, 0);

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}


// =====================================================
// SETUP
// =====================================================

void setup()
{
  Serial.begin(115200);

  // Motor pins
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(ENA, OUTPUT);
  pinMode(ENB, OUTPUT);


  // Encoder pins
  pinMode(LEFT_A, INPUT_PULLUP);
  pinMode(LEFT_B, INPUT_PULLUP);

  pinMode(RIGHT_A, INPUT_PULLUP);
  pinMode(RIGHT_B, INPUT_PULLUP);


  // Encoder interrupts
  attachInterrupt(
    digitalPinToInterrupt(LEFT_A),
    leftEncoderISR,
    CHANGE
  );

  attachInterrupt(
    digitalPinToInterrupt(RIGHT_A),
    rightEncoderISR,
    CHANGE
  );


  stopMotors();

  Serial.println();
  Serial.println("========================================");
  Serial.println(" TWO MOTOR + TWO ENCODER TEST");
  Serial.println("========================================");

  Serial.println();
  Serial.println("Starting in 3 seconds...");
  delay(3000);
}


// =====================================================
// LOOP
// =====================================================

void loop()
{
  static unsigned long startTime = millis();
  static unsigned long lastPrint = 0;

  unsigned long elapsed = millis() - startTime;


  // -----------------------------------------
  // RUN MOTORS FOR 5 SECONDS
  // -----------------------------------------

  if (elapsed < 5000)
  {
    leftMotor(100);
    rightMotor(100);
  }
  else
  {
    stopMotors();
  }


  // -----------------------------------------
  // PRINT ENCODER COUNTS
  // -----------------------------------------

  if (millis() - lastPrint >= 250)
  {
    lastPrint = millis();

    noInterrupts();

    long L = leftCount;
    long R = rightCount;

    interrupts();


    Serial.print("LEFT = ");
    Serial.print(L);

    Serial.print("    RIGHT = ");
    Serial.println(R);
  }
}

Why not VIN → L298N?

The ESP32's VIN/5V pin is meant to supply/receive the board's 5 V rail, not to provide the current needed by two N20 motors.

Two motors can draw substantially more current, especially during startup/stall. If you tried:

ESP32 VIN ───→ L298N motor supply
ESP32 GND ───→ L298N GND

you could get:

ESP32 resets
motor driver not working properly
USB power instability
excessive current through the ESP32/USB supply

And definitely don't connect the motors to the ESP32 3.3 V pin.

If you mean "Can I use the ESP32's VIN/GND just to power the encoder?"

Yes, potentially, if your ESP32 is being powered from USB and its VIN/5V rail is actually at 5 V:

ESP32 3V3 → Encoder VCC
ESP32 GND → Encoder GND

is what we're currently using, though.
