// ================================
// ESP32 SWARM BOT 1 - MOTOR TEST
// ================================

// L298N pins
#define IN1 27
#define IN2 26
#define IN3 25
#define IN4 33

#define ENA 14
#define ENB 32

void leftMotor(int speed) {
  speed = constrain(speed, -255, 255);

  if (speed > 0) {
    digitalWrite(IN1, HIGH);
    digitalWrite(IN2, LOW);
  }
  else if (speed < 0) {
    digitalWrite(IN1, LOW);
    digitalWrite(IN2, HIGH);
  }
  else {
    digitalWrite(IN1, LOW);
    digitalWrite(IN2, LOW);
  }

  analogWrite(ENA, abs(speed));
}

void rightMotor(int speed) {
  speed = constrain(speed, -255, 255);

  if (speed > 0) {
    digitalWrite(IN3, HIGH);
    digitalWrite(IN4, LOW);
  }
  else if (speed < 0) {
    digitalWrite(IN3, LOW);
    digitalWrite(IN4, HIGH);
  }
  else {
    digitalWrite(IN3, LOW);
    digitalWrite(IN4, LOW);
  }

  analogWrite(ENB, abs(speed));
}

void stopMotors() {
  analogWrite(ENA, 0);
  analogWrite(ENB, 0);

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}

void setup() {
  Serial.begin(115200);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(ENA, OUTPUT);
  pinMode(ENB, OUTPUT);

  stopMotors();

  Serial.println();
  Serial.println("==============================");
  Serial.println("   BOT 1 MOTOR TEST");
  Serial.println("==============================");

  delay(2000);
}

void loop() {

  // -----------------------------
  // FORWARD
  // -----------------------------
  Serial.println("FORWARD");

  leftMotor(150);
  rightMotor(150);

  delay(2000);

  stopMotors();
  delay(2000);


  // -----------------------------
  // REVERSE
  // -----------------------------
  Serial.println("REVERSE");

  leftMotor(-150);
  rightMotor(-150);

  delay(2000);

  stopMotors();
  delay(2000);


  // -----------------------------
  // TURN LEFT
  // -----------------------------
  Serial.println("TURN LEFT");

  leftMotor(-150);
  rightMotor(150);

  delay(2000);

  stopMotors();
  delay(2000);


  // -----------------------------
  // TURN RIGHT
  // -----------------------------
  Serial.println("TURN RIGHT");

  leftMotor(150);
  rightMotor(-150);

  delay(2000);

  stopMotors();
  delay(3000);
}

Step 2A — Check Encoder Pulses

Disconnect motor power for now. Keep the ESP32 powered through USB.

Upload this:

#define LEFT_C1   34
#define LEFT_C2   35
#define RIGHT_C1  36
#define RIGHT_C2  39

volatile long leftC1Count  = 0;
volatile long leftC2Count  = 0;
volatile long rightC1Count = 0;
volatile long rightC2Count = 0;

void IRAM_ATTR leftC1ISR() {
  leftC1Count++;
}

void IRAM_ATTR leftC2ISR() {
  leftC2Count++;
}

void IRAM_ATTR rightC1ISR() {
  rightC1Count++;
}

void IRAM_ATTR rightC2ISR() {
  rightC2Count++;
}

void setup() {
  Serial.begin(115200);

  pinMode(LEFT_C1, INPUT);
  pinMode(LEFT_C2, INPUT);
  pinMode(RIGHT_C1, INPUT);
  pinMode(RIGHT_C2, INPUT);

  attachInterrupt(digitalPinToInterrupt(LEFT_C1), leftC1ISR, RISING);
  attachInterrupt(digitalPinToInterrupt(LEFT_C2), leftC2ISR, RISING);
  attachInterrupt(digitalPinToInterrupt(RIGHT_C1), rightC1ISR, RISING);
  attachInterrupt(digitalPinToInterrupt(RIGHT_C2), rightC2ISR, RISING);

  Serial.println();
  Serial.println("==============================");
  Serial.println("       ENCODER TEST");
  Serial.println("==============================");
  Serial.println("Rotate wheels by hand.");
  Serial.println();
}

void loop() {

  static unsigned long lastPrint = 0;

  if (millis() - lastPrint >= 500) {
    lastPrint = millis();

    noInterrupts();

    long lc1 = leftC1Count;
    long lc2 = leftC2Count;
    long rc1 = rightC1Count;
    long rc2 = rightC2Count;

    interrupts();

    Serial.print("LEFT  C1: ");
    Serial.print(lc1);

    Serial.print("   C2: ");
    Serial.print(lc2);

    Serial.print("    |    RIGHT C1: ");
    Serial.print(rc1);

    Serial.print("   C2: ");
    Serial.println(rc2);
  }
}

Why there are two signals per wheel

Each encoder has two channels:

C1 ────▁▁▁▔▔▔▁▁▁▔▔▔───
C2 ─────▁▁▁▔▔▔▁▁▁▔▔▔─

C1 and C2 are slightly shifted in time. This is called quadrature encoding.

That phase difference will later let us determine:

C1 leads C2 → one direction
C2 leads C1 → opposite direction

So eventually the encoder will give us both:

how far the wheel moved + which direction it moved.

For this step, however, we're only checking whether the signals are actually working.

Do not calibrate pulses/revolution yet

First confirm the hardware signals. After that, Step 2B will determine encoder direction and verify that C1/C2 phase changes correctly when you reverse the wheel. Then we'll measure the actual counts per wheel revolution, which we need for the 40 cm movement.
